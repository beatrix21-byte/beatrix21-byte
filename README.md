def guardar(self) -> None:
        """Inserta la inscripción actual en la base de datos."""
        con = obtener_conexion()
        cur = con.cursor(dictionary=True)
        
        query = "INSERT INTO inscripciones (socio_id, clase_id, fecha_inscripcion) VALUES (%s, %s, %s)"
        cur.execute(query, (self.socio_id, self.clase_id, self.fecha_inscripcion))
        
        con.commit()
        cur.close()
        con.close()
